# --- coordinates -------------------------------------------------------------------------
# Latitude/longitude decimals around the study area (lat 33-35, lon 131-134), either order.
latlon-pair	(?<![\d.])3[3-5]\.\d{3,}\D{1,60}?(?<![\d.])13[1-4]\.\d{3,}
lonlat-pair	(?<![\d.])13[1-4]\.\d{3,}\D{1,60}?(?<![\d.])3[3-5]\.\d{3,}
# UTM zone 53N easting/northing near the study area, either order.
utm-pair	(?<![\d.,])[2-4]\d{5}(?:\.\d+)?(?!\d)\D{1,60}?(?<![\d.,])3[78]\d{5}(?:\.\d+)?(?!\d)
utm-pair-rev	(?<![\d.,])3[78]\d{5}(?:\.\d+)?(?!\d)\D{1,60}?(?<![\d.,])[2-4]\d{5}(?:\.\d+)?(?!\d)
# Geometry dumps.
# Requires "x y" (two numbers separated by a space), so prose like "weak point (0.40-0.47)" is not a hit.
wkt-geometry	(?i)\b(?:MULTI)?(?:POINT|LINESTRING|POLYGON)\s*Z?\s*\(\s*\(?\s*\(?\s*-?\d+(?:\.\d+)?\s+-?\d
geojson-coordinates	(?i)"coordinates"\s*:\s*\[

# --- participant / device identifiers ----------------------------------------------------
# Ground-truth stay ids look like {participant}_s{stop}_{device}.
gt-stay-id	(?<![\w])[0-9A-Za-z]{2,6}_s\d{1,3}_[0-9A-Za-z]+
# A participant/device/stay column printed next to a value.
id-value	(?i)\b(?:participant_id|participant|device_id|device|stay_id|user_id|uid)\s*(?:==|=|:)\s*["']?[0-9A-Za-z_-]*\d[0-9A-Za-z_-]*
# Hardware identifiers.
imei-like	(?i)\b(?:imei|serial(?: number| no\.?)?|mac address)\b\D{0,10}[0-9A-F:-]{8,}

# --- local machine, contact details, secrets ---------------------------------------------
windows-path	(?i)\b[A-Z]:[\\/](?:Users|Documents and Settings|Program Files)[\\/]
unix-home-path	(?:^|[\s"'`(=])/(?:home|Users)/[\w.-]+/
onedrive-path	(?i)OneDrive\s*-\s*\w
# The author's own public contact address (site footer) is the one exception, at the user's request (13-10-2026).
email-address	(?<![\w.+-])(?!hafifawwaz@hiroshima-u\.ac\.jp(?![\w.-]))[\w.+-]+@[\w-]+(?:\.[\w-]+)+
github-token	\b(?:ghp|gho|ghu|ghs|ghr)_[A-Za-z0-9]{20,}|\bgithub_pat_[A-Za-z0-9_]{20,}
aws-key	\bAKIA[0-9A-Z]{16}\b
private-key	-----BEGIN [A-Z ]*PRIVATE KEY-----
generic-secret	(?i)\b(?:api[_-]?key|secret|password|passwd|token)\s*[=:]\s*["'][^"']{8,}["']
env-file	(?:^|[\s/\\"'`(])\.env(?![\w-])
local-settings	settings\.local\.json
# Participant codes from the field survey (two digits + group letter A-E, e.g. 01C). Added 13-10-2026:
# an earlier drop-list table carried them unnoticed. Widened 20-10-2026 to also catch code-stop
# forms like 03A-11 (a hyphen followed by a digit), which an inbox note carried past the scanner.
participant-code	(?<![\w.#/-])(?:0[1-9]|1[0-9])[A-E](?!\w|-[^\d])
